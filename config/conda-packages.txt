# build-only-packages.txt are stripped from the payload afterwards.

ros-noetic-compressed-image-transport

# RoboStack's current Noetic builds are for Python 3.12. Pinning it makes the
# solve fail loudly instead of silently falling back to older builds.
python 3.12.*

# --- System libraries (pinned to what RoboStack Noetic builds against) --------
ogre >=1.10.12,<1.11
qt-main 5.15.*
assimp
eigen
yaml-cpp
tinyxml2
urdfdom
urdfdom_headers
libboost-devel
# Keep the payload lean (pulled in indirectly; rviz does not use it):
# numpy defaults to Intel MKL as BLAS on Windows (~540 MiB of DLLs); OpenBLAS
# does the same job in a fraction of the size.
libblas=*=*openblas
# Not pinned: libopencv=*=headless* (drops OpenCV's Qt 6 GUI, ~220 MiB). No
# headless build matches RoboStack's current cv_bridge, so the solver fell back
# to an older RoboStack build set on Python 3.11 (CI run 13).

# --- Build tools (stripped from the payload) ---------------------------------
# Noetic's message generators (genmsg/gencpp/genpy) use the empy 3 API; empy 4
# removed it ("module 'em' has no attribute 'RAW_OPT'"), so pin below 4.
empy >=3.3,<4
cmake >=3.24,<4
ninja
pkg-config
sip
pyqt
pyqt-builder
setuptools
